{{- /* 頁面 bundle 內的影片：靜音自動循環，尊重 prefers-reduced-motion。 用法：{{< local-video src="intro.mp4" poster="intro-poster.jpg" caption="..." >}} src / poster 都是相對於目前 page bundle 的檔名。 */ -}} {{- $src := .Get "src" -}} {{- if not $src -}} {{- errorf "local-video: 缺少 src（%s）" .Page.File.Path -}} {{- end -}} {{- $video := .Page.Resources.GetMatch $src -}} {{- if not $video -}} {{- errorf "local-video: 在 bundle 找不到 %q（%s）" $src .Page.File.Path -}} {{- end -}} {{- $poster := "" -}} {{- with .Get "poster" -}} {{- $p := $.Page.Resources.GetMatch . -}} {{- if $p -}} {{- $poster = $p.RelPermalink -}} {{- else -}} {{- errorf "local-video: 在 bundle 找不到 poster %q（%s）" . $.Page.File.Path -}} {{- end -}} {{- end -}} {{- $caption := .Get "caption" -}}
